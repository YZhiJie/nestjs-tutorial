13. 数据库集成（TypeORM） ​

对应漫画

本章配套四格漫画：EP.13 TypeORM 数据库（含剧情回顾、知识点拆解、坑点清单与自测题）

本章示例：server/src/ch13-database（npm run start:ch13 → http://localhost:3013）

本章说明 ​

Nest 与数据库的集成推荐通过 TypeORM（@nestjs/typeorm）。为了让示例「零外部依赖、开箱即跑」，本章使用 sql.js（WebAssembly 版 SQLite，纯 JS 内存数据库）——无需安装任何数据库，业务代码与生产用 MySQL/PostgreSQL 完全一致，只需替换连接参数。

涉及概念：实体定义、一对多关系、级联保存/删除、Repository 模式、唯一约束处理、启动种子数据。

代码实现 ​

连接配置 ​

ts
// 文件: server/src/ch13-database/app.module.ts
// TypeOrmModule.forRoot：建立数据库连接（教学使用 sql.js 内存库，零外部依赖；
// 生产切换 MySQL/PostgreSQL 只需替换 type 与连接参数，业务代码不变）
import { Module } from '@nestjs/common';
import { TypeOrmModule } from '@nestjs/typeorm';
import * as path from 'node:path';
import { User } from './entities/user.entity';
import { Post } from './entities/post.entity';
import { UsersModule } from './users/users.module';
import { SeedService } from './seed.service';

@Module({
    imports: [
        TypeOrmModule.forRoot({
            type: 'sqljs', // WebAssembly 版 SQLite：纯 JS，无需安装数据库
            sqlJsConfig: {
                // sql.js 需要定位 .wasm 文件（Node 环境按文件系统路径加载）
                locateFile: (file: string) => path.join(process.cwd(), 'node_modules/sql.js/dist', file),
            },
            autoSave: false, // 不持久化到磁盘，进程退出即清空 —— 方便重复演示
            entities: [User, Post],
            synchronize: true, // 按实体自动建表（仅限开发！生产必须用 migrations）
        }),
        TypeOrmModule.forFeature([User]), // SeedService 需要 User 仓库
        UsersModule,
    ],
    providers: [SeedService],
})
export class AppModule {}

forRoot 建立连接，forFeature([User]) 在模块内注册实体仓库。关键参数：

  • type: 'sqljs'：纯 JS 内存库；生产换成 'mysql'/'postgres' + host/port/username/password/database；
  • synchronize: true：按实体自动建表——仅限开发，生产必须关闭并改用 migrations；
  • autoSave: false：sql.js 不落盘，进程退出数据清空，方便重复演示。

实体与关系 ​

ts
// 文件: server/src/ch13-database/entities/user.entity.ts
// TypeORM 实体：类与表结构一一映射（装饰器即建表语句）
import { Column, Entity, OneToMany, PrimaryGeneratedColumn } from 'typeorm';
import { Post } from './post.entity';

@Entity('users') // 表名
export class User {
    @PrimaryGeneratedColumn() // 自增主键
    id!: number;

    @Column({ length: 50 })
    name!: string;

    @Column({ unique: true }) // 唯一约束：重复插入会触发 QueryFailedError
    email!: string;

    @Column({ default: true })
    isActive!: boolean;

    // 一对多：一个用户有多篇文章；cascade 让保存 User 时联动保存 posts
    @OneToMany(() => Post, (post) => post.author, { cascade: true })
    posts!: Post[];
}

// 文件: server/src/ch13-database/entities/post.entity.ts
import { Column, Entity, ManyToOne, PrimaryGeneratedColumn } from 'typeorm';
import { User } from './user.entity';

@Entity('posts')
export class Post {
    @PrimaryGeneratedColumn()
    id!: number;

    @Column({ length: 200 })
    title!: string;

    @Column('text')
    content!: string;

    // 多对一：多篇文章属于一个用户；onDelete: CASCADE 随作者删除联动删除
    @ManyToOne(() => User, (user) => user.posts, { onDelete: 'CASCADE' })
    author!: User;
}

Repository 模式 ​

@InjectRepository(User) 注入实体仓库，CRUD 全部走 Repository API：

ts
// 文件: server/src/ch13-database/users/users.service.ts
// Repository 模式：@InjectRepository 注入实体仓库，全部 CRUD 走 Repository API
import { ConflictException, Injectable, NotFoundException } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { QueryFailedError, Repository } from 'typeorm';
import { User } from '../entities/user.entity';
import { Post } from '../entities/post.entity';
import type { CreateUserDto } from '../dto/create-user.dto';

@Injectable()
export class UsersService {
    constructor(
        @InjectRepository(User) private readonly usersRepo: Repository<User>,
        @InjectRepository(Post) private readonly postsRepo: Repository<Post>,
    ) {}

    findAll(): Promise<User[]> {
        return this.usersRepo.find();
    }

    // relations：查询时 JOIN 加载关联的 posts
    async findOneWithPosts(id: number): Promise<User> {
        const user = await this.usersRepo.findOne({ where: { id }, relations: ['posts'] });
        if (!user) throw new NotFoundException(`用户 ${id} 不存在`);
        return user;
    }

    async create(dto: CreateUserDto): Promise<User> {
        try {
            // create 只构造实体实例，save 才执行 INSERT
            return await this.usersRepo.save(this.usersRepo.create(dto));
        } catch (err) {
            // 唯一约束冲突 → 转成语义明确的 409，而不是把数据库错误抛给客户端
            if (err instanceof QueryFailedError) {
                throw new ConflictException(`邮箱 ${dto.email} 已被注册`);
            }
            throw err;
        }
    }

    async createPost(userId: number, title: string, content: string): Promise<Post> {
        const author = await this.usersRepo.findOneBy({ id: userId });
        if (!author) throw new NotFoundException(`用户 ${userId} 不存在`);
        return this.postsRepo.save(this.postsRepo.create({ title, content, author }));
    }

    async remove(id: number): Promise<void> {
        const user = await this.findOneWithPosts(id);
        await this.usersRepo.remove(user); // posts 随 onDelete: CASCADE 联动删除
    }
}

控制器与 DTO：

ts
// 文件: server/src/ch13-database/users/users.controller.ts
import { Body, Controller, Delete, Get, HttpCode, Param, ParseIntPipe, Post } from '@nestjs/common';
import { UsersService } from './users.service';
import { CreateUserDto } from '../dto/create-user.dto';

@Controller('users')
export class UsersController {
    constructor(private readonly usersService: UsersService) {}

    // GET /users —— 全量用户（不含关联）
    @Get()
    findAll() {
        return this.usersService.findAll();
    }

    // GET /users/1 —— 单用户 + 关联的文章列表
    @Get(':id')
    findOne(@Param('id', ParseIntPipe) id: number) {
        return this.usersService.findOneWithPosts(id);
    }

    // POST /users —— 创建用户；邮箱重复 → 409
    @Post()
    create(@Body() dto: CreateUserDto) {
        return this.usersService.create(dto);
    }

    // POST /users/1/posts —— 为指定用户发布文章
    @Post(':id/posts')
    createPost(
        @Param('id', ParseIntPipe) id: number,
        @Body('title') title: string,
        @Body('content') content: string,
    ) {
        return this.usersService.createPost(id, title, content);
    }

    // DELETE /users/1 —— 删除用户（文章级联删除）
    @Delete(':id')
    @HttpCode(204)
    remove(@Param('id', ParseIntPipe) id: number) {
        return this.usersService.remove(id);
    }
}

启动种子数据 ​

内存库每次启动为空，利用 OnApplicationBootstrap 钩子在启动后写入演示数据：

ts
// 文件: server/src/ch13-database/seed.service.ts
// 启动种子数据：内存数据库每次启动为空，应用启动后写入演示数据
import { Injectable, OnApplicationBootstrap } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository } from 'typeorm';
import { User } from './entities/user.entity';

@Injectable()
export class SeedService implements OnApplicationBootstrap {
    constructor(@InjectRepository(User) private readonly usersRepo: Repository<User>) {}

    async onApplicationBootstrap() {
        const count = await this.usersRepo.count();
        if (count > 0) return;
        // cascade: true 使 posts 随 User 一起保存
        await this.usersRepo.save([
            {
                name: '张三',
                email: 'zhangsan@example.com',
                posts: [
                    { title: 'NestJS 入门', content: '从模块与依赖注入开始……' },
                    { title: 'TypeORM 实战', content: '实体、关系与仓库模式……' },
                ],
            },
            { name: '李四', email: 'lisi@example.com', posts: [] },
        ]);
        console.log('[ch13] 种子数据已写入（2 个用户 / 2 篇文章）');
    }
}

最佳实践与坑点 ​

  1. synchronize: true 仅限开发：它会按实体直接改表结构，生产环境可能导致数据丢失——生产必须关闭并用 typeorm migration 管理结构变更；
  2. create() 与 save() 的分工：create 只在内存构造实体实例，save 才执行 INSERT/UPDATE；
  3. 唯一约束冲突要主动翻译：捕获 QueryFailedError 转 ConflictException（409），别把数据库错误码直接抛给客户端；
  4. sql.js 需要定位 wasm 文件：Node 环境用 locateFile 指向 node_modules/sql.js/dist（见 app.module.ts），换成真实数据库后删除该配置；
  5. 关系的删除方向要想清楚：本章 onDelete: 'CASCADE' 在数据库层联动删除，比应用层手动删干净可靠；
  6. 切换到真实数据库：安装 mysql2 或 pg 驱动 → 修改 forRoot 连接参数 → 业务代码零改动——这正是 ORM 抽象的价值。
